```sql
WITH om AS (SELECT strftime('%Y-%m', order_date) AS month, SUM(amount_cents) AS gross_cents FROM orders GROUP BY 1), rm AS (SELECT strftime('%Y-%m', refund_date) AS month, SUM(amount_cents) AS refunds_cents FROM refunds GROUP BY 1), months AS (SELECT month FROM om UNION SELECT month FROM rm) SELECT m.month, COALESCE(o.gross_cents,0) AS gross_cents, COALESCE(r.refunds_cents,0) AS refunds_cents, COALESCE(o.gross_cents,0)-COALESCE(r.refunds_cents,0) AS net_cents FROM months m LEFT JOIN om o ON o.month=m.month LEFT JOIN rm r ON r.month=m.month WHERE m.month >= '2026-08' AND m.month < '2026-10' ORDER BY m.month
```
